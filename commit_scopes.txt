# Commit Scopes for go-horai
# Use these scopes in conventional commits: type(scope): description
# The scope is optional: omit it for repository-wide changes. Never use `root`.
# Only the first word of a non-comment line is a scope.
# Example: fix(storage/redis): reload scripts after SCRIPT FLUSH

# Repository scopes
deps                        # Dependency updates (Dependabot writes build(deps): bump ...)
README                      # Root README
docs                        # Documentation

# Packages
horai                       # Root package: task state, runs, errors and storage ports
task                        # Registration contract: task configuration and priorities
schedule                    # Schedule parsing and next-run computation
query                       # List-filter expressions over task and history fields
scheduler                   # Engine: dispatch, runs, leases, recovery, management
storage/memory              # In-memory store
storage/sqldb               # PostgreSQL and MySQL/MariaDB store over database/sql
storage/storagetest         # Store contract suite
storage/redis               # Redis store, nested module
storage/mongo               # MongoDB store, nested module
storage/clickhouse          # ClickHouse history store, nested module
horaitest                   # Test doubles for consumers
integration/docker          # Docker-backed integration tests, nested module

# Tooling
devtools                    # Development tool dependencies
ci                          # GitHub Actions workflows and scripts
